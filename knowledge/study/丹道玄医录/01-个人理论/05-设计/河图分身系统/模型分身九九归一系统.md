```python
class NineGridOrchestrator:
    def __init__(self):
        self.agents = self.init_nine_palace_agents()  # 初始化9分身
        self.breakpoint_pool = []                     # 断点池
        self.global_state = {"balance": 1.0, "cycle": 0}  # 全局平衡度

    # 初始化九宫分身
    def init_nine_palace_agents(self):
        return {
            1: NinePalaceAgent(1, "water",  1024, 6144, {}),
            2: NinePalaceAgent(2, "fire",   2048, 7168, {}),
            3: NinePalaceAgent(3, "wood",   3072, 8192, {}),
            4: NinePalaceAgent(4, "wood",   3072, 8192, {}),

            6: NinePalaceAgent(6, "metal",  4096, 9216, {}),
            7: NinePalaceAgent(7, "metal",  4096, 9216, {}),
            8: NinePalaceAgent(8, "earth",  5120, 10240, {}),
            9: NinePalaceAgent(9, "fire",   2048, 7168, {}),
        }

    # ====================== 核心A：Token分片（河图规则） ======================
    def split_token_by_he_tu(self, text, palace_id):
        agent = self.agents[palace_id]
        max_token = agent.token_safe
        # 自动分片，超长文本进入断点续跑
        chunks = [text[i:i+max_token] for i in range(0, len(text), max_token)]
        return chunks

    # ====================== 核心B：洛书循环流转（生生不息） ======================
    def run_luo_shu_cycle(self, task_id, user_input):
        current_palace = LUOSHU_CYCLE[0]  # 从震3宫启动
        remaining_text = user_input
        thought_chain = []

        for palace in LUOSHU_CYCLE:
            agent = self.agents[palace]
            
            # 1. Token分片
            chunks = self.split_token_by_he_tu(remaining_text, palace)
            if not chunks:
                break
            current_chunk = chunks[0]
            remaining_text = "".join(chunks[1:]) if len(chunks) > 1 else ""

            # 2. 执行当前宫位分身推理
            # 感知→解析→拆链→推理→缓存→存储→输出→汇总
            result, state = self.run_agent(palace, current_chunk)
            thought_chain.append({
                "palace": palace,
                "result": result,
                "dantian_state": state
            })

            # 3. 未完成 → 生成断点（断点续跑）
            if remaining_text:
                bp = BreakPointSnap(
                    task_id=task_id,
                    current_palace=palace,
                    element_state=state,
                    thought_steps=thought_chain,
                    remain_prompt=remaining_text,
                    token_used=len(current_chunk),
                    next_palace=LUOSHU_CYCLE[LUOSHU_CYCLE.index(palace)+1]
                )
                self.breakpoint_pool.append(bp)
                return f"任务已分片，断点存储在宫{palace}，下次循环接续"

        # 4. 一轮完成 → 全局反馈校准（反向洛书回流）
        self.apply_global_feedback(thought_chain)
        return {"thought_chain": thought_chain, "global_balance": self.global_state["balance"]}

    # ====================== 核心C：断点续跑 ======================
    def resume_from_breakpoint(self, task_id):
        for bp in self.breakpoint_pool:
            if bp.task_id == task_id:
                # 从断点宫位直接接续
                return self.run_luo_shu_cycle(task_id, bp.remain_prompt)
        return "无断点可续跑"

    # ====================== 核心D：全局自进化反馈 ======================
    def apply_global_feedback(self, thought_chain):
        # 洛书逆序回流修正权重
        error = 0.1  # 模拟误差
        self.global_state["balance"] -= error * 0.2
        for step in thought_chain:
            palace = step["palace"]
            state = step["dantian_state"]
            # 各分身权重微调
            state["balance"] -= error * 0.05
        self.global_state["cycle"] += 1

    # 执行单个分身
    def run_agent(self, palace_id, chunk):
        agent = self.agents[palace_id]
        result = f"【宫{palace_id}·{agent.element}】处理完成：{chunk[:20]}..."
        state = {"element": agent.element, "balance": 0.98, "token_used": len(chunk)}
        return result, state
```

```python
class GlobalFeedback:
    def __init__(self, config):
        self.config = config

    def calibrate(self, thought_chain, global_state):
        error = 0.1
        global_state["balance"] = max(0.5, global_state["balance"] - error * 0.2)
        global_state["cycle"] += 1

        for step in thought_chain:
            state = step.get("dantian_state", {})
            state["balance"] = max(0.6, state.get("balance", 1.0) - error * 0.05)
            state["error_rate"] = error

        return global_state, thought_chain
```

```python
        for pid, cfg in PALACE_TOKEN_CONFIG.items():
            agents[pid] = NinePalaceAgent(
                palace_id=pid,
                element=cfg["element"],
                token_safe=cfg["safe"],
                token_fuse=cfg["fuse"]
            )
        return agents

    def split_token(self, text: str, palace_id: int):
        max_t = self.agents[palace_id].token_safe
        chunks = [text[i:i+max_t] for i in range(0, len(text), max_t)]
        return chunks

    def run_agent(self, palace_id: int, chunk: str):
        agent = self.agents[palace_id]
        result = f"[{palace_id}宫|{agent.element}] 处理：{chunk[:25]}..."
        state = agent.state
        state["token_used"] = len(chunk)
        state["cycle_count"] += 1
        return result, state

    def run_cycle(self, task_id: str, user_input: str):
        remaining = user_input
        chain = []

        for pid in LUOSHU_CYCLE:
            chunks = self.split_token(remaining, pid)
            if not chunks:
                break

            current = chunks[0]
            remaining = "".join(chunks[1:]) if len(chunks) > 1 else ""
            res, state = self.run_agent(pid, current)

            chain.append({
                "palace": pid,
                "element": self.agents[pid].element,
                "result": res,
                "dantian_state": state
            })

            if remaining and self.sys_config.ENABLE_BREAKPOINT_RESUME:
                bp = BreakPointSnap(
                    task_id=task_id,
                    current_palace=pid,
                    element_state=state,
                    thought_chain=chain,
                    remain_text=remaining,
                    token_used=len(current),
                    next_palace=LUOSHU_CYCLE[LUOSHU_CYCLE.index(pid)+1] if (LUOSHU_CYCLE.index(pid)+1 < len(LUOSHU_CYCLE)) else 5,
                    timestamp=time.strftime("%Y-%m-%d %H:%M:%S")
                )
                self.breakpoint_pool.append(bp)
                return {
                    "status": "breakpoint",
                    "palace": pid,
                    "msg": f"已分片断点存储在{pid}宫",
                    "breakpoint": bp
                }

        if self.sys_config.ENABLE_AUTO_FEEDBACK:
            self.global_state, chain = self.feedback.calibrate(chain, self.global_state)

        return {
            "status": "success",
            "task_id": task_id,
            "thought_chain": chain,
            "global_state": self.global_state
        }

    def resume_breakpoint(self, task_id: str):
        for bp in self.breakpoint_pool:
            if bp.task_id == task_id:
                self.breakpoint_pool.remove(bp)
                return self.run_cycle(task_id, bp.remain_text)
        return {"status": "fail", "msg": "无断点"}
```
