```python
class HeartbeatController:
    """
    河图心跳控制器
    - 独立线程运行，优先级全域最高
    - 根据熵值自动切换节拍
    - 永不中断——只改变心跳快慢
    """
    
    def __init__(self, config: HetuoConfig):
        self._config = config
        self._current_tick = config.TICK_NORMAL  # 初始值5s
        self._current_entropy = 0.0
        self._running = False
        self._tick_count = 0
        self._lock = threading.Lock()
    
    def start(self):
        """启动心跳线程"""
        self._running = True
        self._thread = threading.Thread(target=self._loop, daemon=True)
        self._thread.start()
    
    def update_entropy(self, entropy: float):
        """更新全局熵值（由坎宫感知或洛书巡天调用）"""
        with self._lock:
            self._current_entropy = entropy
            self._current_tick = self._compute_tick(entropy)
    
    def _compute_tick(self, entropy: float) -> float:
        """根据熵值自动选择节拍"""
        if entropy >= 0.9:
            return self._config.TICK_CRITICAL    # 1s
        elif entropy >= 0.7:
            return self._config.TICK_ALERT       # 2s
        elif entropy >= 0.3:
            return self._config.TICK_NORMAL      # 5s
        else:
            return self._config.TICK_STEADY      # 10s
    
    def _loop(self):
        """心跳主循环"""
        while self._running:
            with self._lock:
                tick = self._current_tick
            # 触发本轮河图感知（调用现有 LandscapeMapper.update() 或 bridge 接口）
            # ... 
            time.sleep(tick)
```

```python
class QiOmenDetector:
    """
    气象征兆预判器
    核心理念：治未病——在裂隙形成之前就有预感。
    原理：连续监测熵流变化率 d(entropy)/dt，当变化率异常时提前预警。
    """
    
    def __init__(self, window_size: int = 5):
        self._entropy_history: list[float] = []  # 滑动窗口
        self._window_size = window_size
        self._trend: str = "stable"  # stable | rising | falling | volatile
        self._omen_score: float = 0.0
    
    def feed(self, entropy: float) -> dict:
        """
        灌入当前熵值，输出预判结果。
        
        Returns:
            {"trend": "rising", "omen_score": 0.65, 
             "prediction": "未来1-3轮可能达到中度失衡",
             "suggested_action": "提前准备洛书巡天"}
        """
        self._entropy_history.append(entropy)
        if len(self._entropy_history) > self._window_size:
            self._entropy_history.pop(0)
        
        if len(self._entropy_history) < 3:
            return {"trend": "stable", "omen_score": 0.0, 
                    "prediction": "数据不足", "suggested_action": "继续监测"}
        
        # 计算一阶导数（变化率）
        deltas = [self._entropy_history[i+1] - self._entropy_history[i] 
                  for i in range(len(self._entropy_history)-1)]
        avg_delta = sum(deltas) / len(deltas)
        
        # 计算二阶导数（加速度）
        if len(deltas) >= 2:
            accels = [deltas[i+1] - deltas[i] for i in range(len(deltas)-1)]
            avg_accel = sum(accels) / len(accels)
        else:
            avg_accel = 0.0
        
        # 趋势判定
        if abs(avg_delta) < 0.02:
            self._trend = "stable"
        elif avg_delta > 0 and avg_accel > 0:
            self._trend = "accelerating_rise"  # 加速上升——危险
        elif avg_delta > 0:
            self._trend = "rising"
        elif avg_delta < 0 and avg_accel < 0:
            self._trend = "accelerating_fall"  # 加速下降——可能过大修正
        else:
            self._trend = "falling"
        
        # 气象征兆评分（0-1）
        self._omen_score = min(1.0, abs(avg_delta) * 3 + abs(avg_accel) * 5)
        
        return {
            "trend": self._trend,
            "omen_score": round(self._omen_score, 3),
            "prediction": self._generate_prediction(),
            "suggested_action": self._suggest_action(),
        }
    
    def _generate_prediction(self) -> str:
        if self._omen_score > 0.8:
            return "未来1-2轮可能触发重度失衡"
        elif self._omen_score > 0.5:
            return "未来1-3轮可能达到中度失衡"
        elif self._omen_score > 0.2:
            return "未来2-5轮可能出现轻度波动"
        return "近期稳态可维持"
    
    def _suggest_action(self) -> str:
        if self._omen_score > 0.6:
            return "提前预热洛书巡天调度器"
        elif self._omen_score > 0.3:
            return "加强坎宫感知采样频率"
        return "继续监测"
```

```python
class LuoshuScheduler:
    """
    洛书九宫DAG调度器
    
    执行方式（改良）：
    坎1 ──→ 坤2 ──→ 震3 ──→ 巽4 ──→ 中5 ──→ 乾6 ──→ 兑7 ──→ 艮8 ──→ 离9
      │                  ↓
      └──→ 另线程预加载   └──→ 路径A：坎→坤→震→巽
                                      ↓
                              路径B：中→乾→兑→艮→离
                                      ↓
                              路径C：最终串行归总（离9）
    
    三管齐下：
    - 路径A（数据链路）：坎1 + 坤2 + 震3 + 巽4（并行）
    - 路径B（架构链路）：中5 + 乾6 + 兑7 + 艮8 + 离9（并行）
    - 路径C（归总）：离9 等待 A/B 完成后再执行最终校验
    """
    
    def __init__(self, mapper, config, omen_detector):
        self._mapper = mapper
        self._config = config
        self._omen = omen_detector
        self._feixing = AdaptiveFeixing(config)
        self._arbiter = GrayZoneArbiter()
        self._seeds = MemorySeedManager()
        self._consistency = SelfConsistencyChecker()
    
    def scan(self, ninepalace: dict) -> dict:
        """
        执行一轮洛书巡天。
        
        Returns:
            fissure_list, global_entropy, proposal(如有)
        """
        # 1. 先查记忆种子——是否有同类问题的最优路径
        seed = self._seeds.query(ninepalace)
        if seed and self._validate_seed(seed, ninepalace):
            # 种子模式：复用路径 + 微调
            return self._execute_seed(seed, ninepalace)
        
        # 2. 无种子：全量飞星
        # 2a. 自适应跳过稳态宫位
        scan_plan = self._feixing.build_plan(ninepalace)
        
        # 2b. DAG 并行执行
        with ThreadPoolExecutor(max_workers=2) as executor:
            future_a = executor.submit(self._scan_path_a, ninepalace, scan_plan)
            future_b = executor.submit(self._scan_path_b, ninepalace, scan_plan)
            result_a = future_a.result()
            result_b = future_b.result()
        
        # 2c. 离9归总校验
        final = self._merge_paths(result_a, result_b, ninepalace)
        
        # 3. 灰色地带仲裁
        final["anomalies"] = self._arbiter.filter(final["anomalies"])
        
        # 4. 自洽性校验
        if final.get("proposal"):
            final["proposal_valid"] = self._consistency.check(final["proposal"])
        
        # 5. 保存记忆种子（用于下次复用）
        if final.get("fissure_list"):
            self._seeds.save(self._build_seed(final, ninepalace))
        
        return final
```

```python
class LayeredApproval:
    """
    三级审批引擎
    
    层级                   风险评分范围    审批方式           典型耗时
    一级·自动授权         < 0.3          自动放行           即时
    二级·简化审批       0.3 ~ 0.6       中宫摘要推送 + 60min等待   ≤60min
    三级·完整审批        > 0.6          完整提案 + 24h等待         ≤24h
    """
    
    RISK_DIMENSIONS = {
        "影响范围": {"权重": 0.4, ...},
        "可回滚性": {"权重": 0.3, ...},
        "与用户领域匹配度": {"权重": 0.3, ...},
    }
    
    APPROVAL_LEVELS = [
        {"name": "auto", "max_risk": 0.3, "timeout": 0},
        {"name": "simplified", "max_risk": 0.6, "timeout": 3600},
        {"name": "full", "max_risk": 1.0, "timeout": 86400},
    ]
    
    def compute_risk(self, proposal: dict) -> float:
        """计算风险评分"""
        score = 0.0
        score += self._score_scope(proposal) * 0.4      # 影响范围
        score += self._score_rollback(proposal) * 0.3    # 可回滚性
        score += self._score_domain(proposal) * 0.3       # 领域匹配
        return min(1.0, score)
    
    def submit(self, proposal: dict) -> dict:
        """提交提案到审批队列"""
        risk = self.compute_risk(proposal)
        level = self._determine_level(risk)
        
        proposal["risk_score"] = risk
        proposal["approval_level"] = level["name"]
        proposal["status"] = "pending"
        proposal["submitted_at"] = time.time()
        proposal["deadline"] = time.time() + level["timeout"]
        
        if level["name"] == "auto":
            proposal["status"] = "approved"
            proposal["approved_at"] = time.time()
            return proposal  # 立即放行
        
        # 发送到审批队列（需用户确认）
        # ...
        return proposal
```

```python
class StrategyOptimizer:
    """
    策略自优化器
    核心能力：系统越用越适配——根据历史数据自动调优五宫权重。
    安全约束：所有优化参数受中宫终审兜底，用户可一键恢复出厂。
    """
    
    def __init__(self, config: HetuoConfig, 
                 adjustment_window: int = 50):  # 每50轮tick评估一次
        self._config = config
        self._adjustment_window = adjustment_window
        self._tick_counter = 0
        # 五宫效能历史
        self._palace_performance: dict[str, list[float]] = {
            "kan": [], "kun": [], "zhen": [], "xun": [],
            "zhong": [], "qian": [], "dui": [], "gen": [], "li": []
        }
    
    def record_palace_performance(self, palace: str, entropy_delta: float, 
                                   time_cost: float):
        """记录单宫效能"""
        efficiency = entropy_delta / max(time_cost, 0.001)
        self._palace_performance[palace].append(efficiency)
        # 保持窗口大小
        if len(self._palace_performance[palace]) > self._adjustment_window:
            self._palace_performance[palace].pop(0)
    
    def evaluate_and_adjust(self) -> dict | None:
        """每50轮评估一次，决定是否调整策略"""
        self._tick_counter += 1
        if self._tick_counter % self._adjustment_window != 0:
            return None
        
        adjustments = {}
        for palace, history in self._palace_performance.items():
            if len(history) < 10:
                continue
            avg_eff = sum(history) / len(history)
            recent_eff = sum(history[-10:]) / 10
            
            # 最近10轮效能明显下降 → 建议调整
            if recent_eff < avg_eff * 0.7:
                adjustments[palace] = {
                    "action": "increase_sampling",  # 建议提高采样频率
                    "reason": f"效能下降{(1-recent_eff/avg_eff)*100:.0f}%"
                }
        
        return adjustments if adjustments else None
```

```
□ 策略自优化器每50轮正确输出调整建议
□ 调整建议不自动执行——需中宫审批（安全约束）
□ 反馈学习循环正确更新基线熵值
```
